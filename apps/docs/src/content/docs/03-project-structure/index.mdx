# Структура проекта FDA

---

## 🗂️ Корневая структура

```
project/
├── apps/                    # Приложения (Astro, SvelteKit, React)
│   ├── docs/               # Документация (Astro)
│   └── web/                # Веб-приложение
│
├── packages/               # Общие пакеты (mono-repo)
│   ├── ui/                 # UI библиотека (Button, Input, etc.)
│   ├── utils/              # Общие утилиты
│   └── config/             # Конфиги (ESLint, Prettier)
│
├── src/                    # Общие исходники (если нет mono-repo)
│
├── package.json
├── pnpm-workspace.yaml     # Для pnpm mono-repo
└── README.md
```

---

## 📁 `src/` — исходники приложения

### Корневая структура:

```
src/
├── lib/                    # Framework-agnostic код
│   ├── server/             # Серверные утилиты и repo
│   ├── universal/          # Общий код для сервера и клиента
│   └── stores/             # Глобальное состояние
│
├── routes/                 # Framework-specific pages
│
└── app.*                   # Точки входа (app.html, app.ts)
```

## 🔧 `src/lib/` — Framework-agnostic код

### Структура:

```
src/lib/
├── server/                 # Серверные утилиты и репозитории
│   ├── repo/               # Внешние источники данных
│   │   ├── db/            # База данных (Drizzle ORM)
│   │   ├── redis/         # Кэш и сессии
│   │   ├── vault/         # Секреты и credentials
│   │   └── keto/          # Авторизация (OpenFGA)
│   ├── utils/             # Серверные утилиты
│   └── db/                # Драйверы БД
│
├── universal/             # Код, работающий и на сервере, и на клиенте
│   ├── stores/            # Глобальное состояние (zustand-like)
│   ├── utils/             # Универсальные утилиты
│   └── types/             # Общие типы
│
└── stores/                # Альтернативное расположение stores
    ├── navigation.ts
    ├── user.ts
    └── theme.ts
```

### Подробное описание:

#### `lib/server/repo/` — Внешние источники данных

```
src/lib/server/repo/
├── db/                     # Работа с базой данных
│   ├── client.ts          # Клиент Drizzle ORM
│   ├── schema.ts          # Схемы таблиц (только Drizzle)
│   └── operations.ts      # Общие операции
│
├── redis/                  # Кэш и сессии
│   ├── client.ts          # Клиент Redis
│   └── cache.ts           # Кеширование
│
├── vault/                  # Управление секретами
│   ├── client.ts          # Клиент Vault
│   └── secrets.ts         # Работа с секретами
│
└── keto/                   # Авторизация (OpenFGA)
    ├── client.ts          # Клиент Keto
    ├── policy.ts          # Правила доступа
    └── check.ts           # Проверка прав
```

#### `lib/universal/stores/` — Глобальное состояние

```
src/lib/universal/stores/
├── navigation.ts          # Навигация (текущий путь, history)
├── user.ts                # Информация о пользователе
├── theme.ts               # Тема (light/dark mode)
├── toast.ts               # Оповещения (для useToast)
└── modal.ts               # Модальные окна
```

## 🏗️ Детальная структура модуля

Каждый модуль (например, `dashboard/`) имеет следующую структуру:

```
dashboard/
├── +layout.svelte         # Обязательно
├── +layout.server.ts      # Опционально
├── +page.svelte           # Обязательно (страница модуля)
├── +page.server.ts        # Опционально (actions, load)
│
├── rpc.ts                 # Обязательно — API для других модулей
├── model.server.ts        # Обязательно — логика работы с данными
├── controller.ts          # Обязательно — управление состоянием
│
├── policy.ts              # Опционально — авторизация
├── utils.ts               # Опционально — утилиты модуля
├── types.ts               # Опционально — типы
├── templates.ts           # Опционально — шаблоны форм
```

#### `+layout.server.ts` — Load для layout

```typescript
// src/routes/dashboard/+layout.server.ts
import { redirect } from '@sveltejs/kit'
import type { LayoutLoad } from './$types'

export const load: LayoutLoad = async ({ parent }) => {
  const { user } = await parent()
  
  if (!user) {
    throw redirect(302, '/login')
  }
  
  return { title: 'Dashboard' }
}
```

#### `+page.server.ts` — Actions и load

```typescript
// src/routes/dashboard/+page.server.ts
import { db } from '$lib/server/repo/db/client'
import { users } from '$lib/server/repo/db/schema'
import { eq } from 'drizzle-orm'
import { fail } from '@sveltejs/kit'
import type { Actions, PageServerLoad } from './$types'

export const load: PageServerLoad = async ({ locals }) => {
  const user = await db.query.users.findFirst({
    where: eq(users.id, locals.user.id),
  })
  
  return { user }
}

export const actions: Actions = {
  updateProfile: async ({ request, locals }) => {
    const formData = await request.formData()
    const name = formData.get('name') as string
    
    await db
      .update(users)
      .set({ name })
      .where(eq(users.id, locals.user.id))
    
    return { success: true }
  },
}. satisfies Actions
```

#### `rpc.ts` — API между модулями

```typescript
// src/routes/dashboard/rpc.ts
import { db } from '$lib/server/repo/db/client'
import { users, posts } from '$lib/server/repo/db/schema'
import { eq } from 'drizzle-orm'
import type { Rpc } from '$lib/universal/types'

export const getDashboardStats: Rpc<'dashboard/getStats'> = async ({ userId }) => {
  const [userStats, postStats] = await Promise.all([
    db.selectFrom('users').where('id', '=', userId).selectAll(),
    db.selectFrom('posts')
      .where('authorId', '=', userId)
      .select({ count: count() })
      .executeTakeFirst(),
  ])
  
  return { userStats, postStats }
}
```

#### `model.server.ts` — Логика работы с данными

```typescript
// src/routes/dashboard/model.server.ts
import { db } from '$lib/server/repo/db/client'
import { users, posts, userStats } from '$lib/server/repo/db/schema'
import { eq, and } from 'drizzle-orm'
import type { DashboardStats } from './types'

export const getDashboardData = async (userId: string): Promise<DashboardStats> => {
  const [user, postsCount, commentsCount] = await Promise.all([
    db.query.users.findFirst({
      where: eq(users.id, userId),
      columns: { id: true, name: true, email: true },
    }),
    db.selectFrom('posts')
      .where(eq(posts.authorId, userId))
      .select({ count: count() })
      .executeTakeFirst(),
    db.selectFrom('comments')
      .where(eq(posts.authorId, userId))
      .select({ count: count() })
      .executeTakeFirst(),
  ])
  
  return { user, postsCount, commentsCount }
}
```

#### `controller.ts` / `stores.ts` — Управление состоянием

```typescript
// src/routes/dashboard/controller.ts
import { derived, writable, type Writable } from 'svelte/store'
import { rpc } from '$lib/universal/rpc'
import type { DashboardStats } from './types'

export const statsStore: Writable<DashboardStats | null> = writable(null)
export const isLoading = writable(false)

export const loadStats = async (userId: string) => {
  isLoading.set(true)
  try {
    const stats = await rpc('dashboard/getStats', { userId })
    statsStore.set(stats)
  } finally {
    isLoading.set(false)
  }
}

export const formattedStats = derived(statsStore, ($stats) => {
  return $stats
    ? {
        posts: $stats.postStats.count,
        comments: $stats.commentStats.count,
      }
    : null
})
```

#### `policy.ts` — Авторизация

```typescript
// src/routes/dashboard/policy.ts
import type { User } from '$lib/universal/types'

export const canViewDashboard = (user: User | null): boolean => {
  return !!user && user.role === 'admin'
}

export const canEditPost = (user: User, postId: string): boolean => {
  return user.role === 'admin' || user.id === postId
}

export const policy = {
  canViewDashboard,
  canEditPost,
}
```
